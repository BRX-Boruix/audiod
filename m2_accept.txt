
    /// M2 acceptance: the EXACT two-tone scenario the end-to-end run performs.
    ///
    /// Stream A writes +16384, stream B writes -16383, both to stereo. Summing and
    /// applying the 1/N gain must yield +0.5 in f32 and 16383 after conversion.
    ///
    /// This test exists because the two streams in the running system write
    /// DISTINGUISHABLE content by design. Two streams writing the SAME sequence would
    /// sum and rescale back to exactly that sequence, making the mixed output
    /// indistinguishable from a single stream at unity gain -- the test would pass
    /// even if only one path were connected. Opposite signs make every failure mode
    /// produce a clearly different number:
    ///   both paths  -> +0.5   -> 16383 or 16384
    ///   only A      -> +1.0   -> 32767 (saturated)
    ///   only B      -> -1.0   -> -32768 (saturated)
    ///   neither     -> silence, which the caller detects as zero bytes written
    #[test]
    fn test_m2_two_tone_acceptance_matches_end_to_end_scenario() {
        const AMP_HI: i16 = 16384;
        const AMP_LO: i16 = -16383;

        // Stereo duplication of each mono tone, as the producers write it.
        let a = mono_to_stereo(&[s16_to_f32(AMP_HI), s16_to_f32(AMP_HI)]);
        let b = mono_to_stereo(&[s16_to_f32(AMP_LO), s16_to_f32(AMP_LO)]);

        let mixed = mix_add(&[&a, &b]);
        for (i, s) in mixed.iter().enumerate() {
            assert!(
                (s - 0.5).abs() < 1e-4,
                "sample {i}: expected +0.5 from (16384 - 16383)/2, got {s}"
            );
            assert!(
                !(-1.0..=1.0).contains(s) || s.abs() <= 1.0,
                "sample {i} escaped [-1, 1]: {s}"
            );
        }

        // Matches what the DMA dump should show: about +0.5 full scale.
        let s16 = f32_to_s16(mixed[0]);
        assert!(
            (s16 - 16383).abs() <= 2,
            "expected about 16383 in s16, got {s16}"
        );

        // Negative control: a single path at unity gain must NOT look like the mix.
        let only_a = mix_add(&[&a]);
        assert!(
            f32_to_s16(only_a[0]) > 32000,
            "single path must saturate high, proving the mix is distinguishable"
        );
    }
